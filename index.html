WITH DedupedTransactions AS (

    SELECT

        transaction_id,

        patient_id,

        service_category,

        CAST(transaction_timestamp AS DATE) AS tx_date,

        COALESCE(invoice_amount, 0.0) AS gross_revenue,

        COALESCE(clinic_cost, 0.0) AS direct_cost,

        ROW_NUMBER() OVER (

            PARTITION BY transaction_id

            ORDER BY updated_at DESC

        ) AS rn

    FROM raw_clinic_sales

    WHERE transaction_status = 'COMPLETED'

),

MonthlyRollup AS (

    SELECT

        DATE_TRUNC('month', tx_date) AS reporting_month,

        COUNT(DISTINCT patient_id) AS active_patients,

        SUM(gross_revenue) AS monthly_revenue,

        SUM(gross_revenue - direct_cost) AS net_gross_margin,

        ROUND(SUM(gross_revenue - direct_cost) / NULLIF(SUM(gross_revenue), 0) * 100, 2) AS margin_pct

    FROM DedupedTransactions

    WHERE rn = 1

    GROUP BY 1

)

SELECT * FROM MonthlyRollup ORDER BY reporting_month ASC;
